\section{4 --- Our v8 vs the AI version}

\begin{frame}{Two different v8 designs}
    Group~2 ended up with \textbf{two} v8s. They take opposite bets.

    \vspace{0.5em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            {\small\textbf{Our v8}}\\[0.25em]
            {\footnotesize
            Starts from v7 and keeps a
            \textbf{thread-cache-first} hot path.
            Makes the cache smarter: higher
            per-class cap (512), live-block
            accounting, retune every 256 allocs,
            occupied-bin bitmaps, optional metrics
            and build scripts
            (\texttt{UsingMatrixClassFork}, branch
            \texttt{v8}).
            }
        \end{column}
        \begin{column}{0.48\textwidth}
            {\small\textbf{The AI version}}\\[0.25em]
            {\footnotesize
            AI-assisted redesign aimed at
            \textbf{packed matrix rows}.
            Turns the hot-path cache \emph{off},
            splits the arena, 64-aligns medium
            rows, page-aligns only at 2\,MiB,
            keeps exact remainders.
            This is what we profiled on the
            M3~Max matrix campaign.
            }
        \end{column}
    \end{columns}

    \vspace{0.55em}
    {\footnotesize Result tables below are for \textbf{the AI version}
    unless we say otherwise.}
\end{frame}

\begin{frame}{Our v8 --- what changed from v7}
    {\scriptsize
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{@{}p{0.34\textwidth}p{0.58\textwidth}@{}}
        \toprule
        Change & Effect \\
        \midrule
        Max cache / class \(256\rightarrow512\) &
          Larger per-class LIFO retention \\
        Live-block accounting &
          Track live blocks per size class \\
        Retune every 256 allocs &
          Raise/lower targets from peak live use
          (cushion 4; min 8) \\
        Occupied-bin bitmaps &
          Skip empty size classes in search \\
        Optional metrics + scripts &
          Hits/misses, refills, flushes, mutex takes;
          build/run helpers \\
        \bottomrule
    \end{tabular}
    }

    \vspace{0.45em}
    {\footnotesize Big policy unchanged: hot path still uses
    \texttt{takeCachedBlock}; \texttt{std::mutex}; page-align @ 64\,KiB.}
\end{frame}

\begin{frame}{Our v8 vs the AI version}
    {\scriptsize
    \renewcommand{\arraystretch}{1.45}
    \setlength{\tabcolsep}{4pt}
    \begin{tabular}{@{}p{0.20\textwidth}p{0.36\textwidth}p{0.36\textwidth}@{}}
        \toprule
        & \textbf{Our v8} & \textbf{The AI version} \\
        \midrule
        Goal &
          Smarter tiny-object cache &
          Packed matrix rows \\
        Hot path &
          Thread cache (kept) &
          Cache skipped \\
        Arena &
          One mmap region &
          Small + guard + large \\
        Cache cap &
          Up to 512 / class &
          Not used on hot path \\
        Align &
          Page @ 64\,KiB &
          64-byte @ \(\ge\)2\,KiB; page @ 2\,MiB \\
        Locking &
          \texttt{std::mutex} &
          \texttt{pthread} + recent-large \\
        Best for &
          Many small allocs &
          Reused medium rows \\
        \bottomrule
    \end{tabular}
    }
\end{frame}

\begin{frame}{The AI version --- matrix packing (and list cost)}
    {\scriptsize
    \renewcommand{\arraystretch}{1.35}
    \begin{tabular}{@{}p{0.30\textwidth}p{0.62\textwidth}@{}}
        \toprule
        Change & Why \\
        \midrule
        Skip thread cache & LIFO cache scatters packed rows \\
        Split arena + guard & Tiny frees cannot punch row region \\
        64-byte align \(\ge\)2\,KiB & Double rows without page waste \\
        Page-align only @ 2\,MiB & 64\,KiB page-align wastes KiB rows \\
        Exact remainders & Packed stride between rows \\
        \bottomrule
    \end{tabular}
    }

    \vspace{0.4em}
    \textbf{Lists break:} millions of tiny nodes + cache off \(\Rightarrow\)
    locked best-fit every time; 16\,MiB small arena spills.

    \vspace{0.3em}
    {\footnotesize Pattern-optimized (profiled \(n{=}360\)), not a hardcoded size.
    Our v8's cache path is the opposite bet.}
\end{frame}
