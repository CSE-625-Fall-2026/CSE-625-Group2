\section{Our v8 vs the AI version}

\begin{frame}{Two different v8 designs}
    Group~2 has \textbf{two} v8s. They take opposite bets.

    \vspace{0.55em}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            {\large\textbf{Our v8}}

            \vspace{0.35em}
            {\footnotesize
            Starts from v7 and keeps a
            \textbf{thread-cache-first} hot path.
            Makes the cache smarter: higher
            per-class cap (512), live-block
            accounting, retune every 256 allocs,
            occupied-bin bitmaps, optional metrics
            and build scripts
            (\texttt{UsingMatrixClassFork} /
            branch \texttt{v8}).
            }
        \end{column}
        \begin{column}{0.48\textwidth}
            {\large\textbf{The AI version}}

            \vspace{0.35em}
            {\footnotesize
            AI-assisted redesign for
            \textbf{packed matrix rows}.
            Turns the hot-path cache \emph{off},
            splits the arena, 64-aligns medium
            rows, page-aligns only at 2\,MiB,
            keeps exact remainders.
            This is what we profiled on the
            M3~Max matrix campaign.
            }
        \end{column}
    \end{columns}

    \vspace{0.7em}
    {\footnotesize Result tables later are for \textbf{the AI version}
    unless we say otherwise.}
\end{frame}

    {\footnotesize Our v8 = better tiny-object cache.

\end{frame}

\begin{frame}{Why the AI version helps matrix / hurts lists}
    \textbf{Matrix win}
    \begin{itemize}
        \item Many reused medium rows (profiled \(n{=}360\),
              \(\sim\)2880\,B/row --- pattern, not a hardcoded size)
        \item No LIFO cache \(\Rightarrow\) rows stay packed
        \item Split arena keeps tiny frees out of the row region
    \end{itemize}

    \vspace{0.45em}
    \textbf{List failure}
    \begin{itemize}
        \item Millions of tiny nodes + cache off \(\Rightarrow\)
              lock + bin walk on every \texttt{new}/\texttt{delete}
        \item 16\,MiB small arena fills and spills into the large region
        \item Deferred coalesce becomes the common path
    \end{itemize}

    \vspace{0.4em}
    {\footnotesize Our v8's cache path is the opposite bet --- better for
    tiny objects, not a packing win.}
\end{frame}
