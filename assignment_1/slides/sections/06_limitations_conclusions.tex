\section{6 --- Pthreads \& wrap-up}

\begin{frame}{Pthreads profiling}
    {\footnotesize \(n{=}800\), 8 threads, 200 matrices.
    pthreads \textbf{v4} = queues/mutexes (simpler);
    \textbf{v6} = less locking (faster, harder to read).
    Each with CMA \textbf{v7} or regular \texttt{new}.}

    \vspace{0.3em}
    {\scriptsize
    \begin{tabular}{@{}lrr@{}}
        \toprule
        Method & WSL mean (s) & Linux mean (s) \\
        \midrule
        v6 + regular & \textbf{292.8} & \textbf{224.9} \\
        v6 + CMA v7 & 300.9 & 226.8 \\
        v4 + regular & 395.8 & 348.6 \\
        v4 + CMA v7 & 397.7 & \textbf{247.0} \\
        \bottomrule
    \end{tabular}
    }

    \vspace{0.5em}
    \begin{itemize}
        \item \textbf{WSL:} CMA slightly worse; v6+regular is best
        \item \textbf{Linux:} CMA \textbf{rescues} v4 (\(348{\rightarrow}247\));
              v6 \(\pm\) CMA not significantly different
        \item Scaling \(n{=}400\)--\(1800\): \(\approx\)cubic;
              v4+CMA still \(8.5\%\)--\(38\%\) slower than v6+regular in range
        \item Lesson: environment changes whether CMA helps threads
    \end{itemize}
\end{frame}

\begin{frame}{Takeaways}
    \begin{enumerate}
        \item CMA vs regular is \textbf{workload-dependent}
        \item On our matrix campaign: \textbf{v1/v2 \(>\) regular \(>\) v5/v7}
        \item \textbf{v8} \(\approx\!-5\%\) vs regular by packing rows ---
              not a general-purpose malloc
        \item Linked lists need the opposite policy (tiny-object cache)
        \item Pthreads: on Linux, CMA can make simpler code competitive;
              WSL tells a different story
    \end{enumerate}

\end{frame}
