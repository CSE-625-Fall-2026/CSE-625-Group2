\section{6 --- Pthreads \& wrap-up}

\begin{frame}{Pthreads profiling}
    {\footnotesize \(n{=}800\), 8 threads, 200 matrices.
    pthreads \textbf{v4} = queues/mutexes (simpler);
    \textbf{v6} = less locking (faster, harder to read).
    Each with CMA \textbf{v7} or regular \texttt{new}.}

    \vspace{0.3em}
    {\scriptsize
    \begin{tabular}{@{}lrr@{}}
        \toprule
        Method & WSL mean (s) & Linux mean (s) \\
        \midrule
        v6 + regular & \textbf{292.8} & \textbf{224.9} \\
        v6 + CMA v7 & 300.9 & 226.8 \\
        v4 + regular & 395.8 & 348.6 \\
        v4 + CMA v7 & 397.7 & \textbf{247.0} \\
        \bottomrule
    \end{tabular}
    }

    \vspace{0.5em}
    \begin{itemize}
        \item \textbf{WSL:} CMA slightly worse; v6+regular is best
        \item \textbf{Linux:} CMA \textbf{rescues} v4 (\(348{\rightarrow}247\));
              v6 \(\pm\) CMA not significantly different
        \item Lesson: environment changes whether CMA helps threads
    \end{itemize}
\end{frame}

\begin{frame}{Takeaways (from the latest report)}
    \begin{enumerate}
        \item No CMA version was faster everywhere --- \textbf{workload decided}
        \item Newer \(\neq\) better: on our matrix campaign,
              \textbf{v1/v2 \(>\) regular \(>\) v5/v7}
        \item CMA trades memory for fewer page faults / less kernel time
        \item \textbf{v8} wins its matrix target (\(\approx\!-5\%\) on E1) and
              fails on lists --- pattern-tuned, not general-purpose
        \item Environment matters (machines, WSL vs Linux, args)
    \end{enumerate}

\end{frame}
